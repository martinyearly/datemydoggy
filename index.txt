function dogSVG(cfg){
  const {fur, spotColor, ear, tongue, cheek} = cfg;
  let ears = '';
  if(ear === 'floppy'){
    ears = `
      <ellipse cx="26" cy="46" rx="15" ry="22" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" transform="rotate(-18 26 46)"/>
      <ellipse cx="74" cy="46" rx="15" ry="22" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" transform="rotate(18 74 46)"/>`;
  } else if(ear === 'pointy'){
    ears = `
      <polygon points="28,50 18,14 42,42" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" stroke-linejoin="round"/>
      <polygon points="72,50 82,14 58,42" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" stroke-linejoin="round"/>`;
  } else {
    ears = `
      <path d="M22 48 Q14 22 34 30 Q30 44 22 48 Z" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" stroke-linejoin="round"/>
      <path d="M78 48 Q86 22 66 30 Q70 44 78 48 Z" fill="${fur}" stroke="#1E2A45" stroke-width="2.5" stroke-linejoin="round"/>`;
  }
  const spots = spotColor ? `
      <ellipse cx="34" cy="42" rx="8" ry="7" fill="${spotColor}" opacity="0.85"/>
      <ellipse cx="66" cy="60" rx="6" ry="5" fill="${spotColor}" opacity="0.85"/>` : '';
  const tongueEl = tongue ? `<path d="M46 78 Q50 92 54 78 Q50 84 46 78 Z" fill="#FF8FAB" stroke="#1E2A45" stroke-width="2"/>` : '';
  const cheekEl = cheek ? `
      <circle cx="34" cy="66" r="6" fill="${cheek}" opacity="0.55"/>
      <circle cx="66" cy="66" r="6" fill="${cheek}" opacity="0.55"/>` : '';

  return `
  <svg viewBox="0 0 100 100" xmlns="http://www.w3.org/2000/svg">
    <rect width="100" height="100" fill="#FFF3D6"/>
    ${ears}
    <circle cx="50" cy="56" r="30" fill="${fur}" stroke="#1E2A45" stroke-width="2.5"/>
    ${spots}
    ${cheekEl}
    <ellipse cx="50" cy="66" rx="15" ry="12" fill="#FFFDF6" stroke="#1E2A45" stroke-width="2"/>
    <circle cx="40" cy="50" r="4" fill="#1E2A45"/>
    <circle cx="60" cy="50" r="4" fill="#1E2A45"/>
    <circle cx="41.3" cy="48.5" r="1.2" fill="#fff"/>
    <circle cx="61.3" cy="48.5" r="1.2" fill="#fff"/>
    <ellipse cx="50" cy="64" rx="4.5" ry="3.4" fill="#1E2A45"/>
    <path d="M50 67 Q50 74 44 74" stroke="#1E2A45" stroke-width="2" fill="none" stroke-linecap="round"/>
    <path d="M50 67 Q50 74 56 74" stroke="#1E2A45" stroke-width="2" fill="none" stroke-linecap="round"/>
    ${tongueEl}
  </svg>`;
}

let toastTimer;
function sendSniff(name){
  const toast = document.getElementById('toast');
  toast.textContent = `Sniff sent to ${name}! You'll hear a bark if it's mutual. 🐾`;
  toast.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => toast.classList.remove('show'), 3200);
}

/* ---------------- Router ---------------- */
function navigate(path){
  document.querySelector('nav.links').classList.remove('open');
  if(path.startsWith('#')){
    if(location.hash !== '#/'){ location.hash = '#/'; }
    requestAnimationFrame(() => {
      const el = document.getElementById(path.slice(1));
      if(el) el.scrollIntoView({behavior:'smooth', block:'start'});
    });
    return;
  }
  location.hash = '#/' + path;
  window.scrollTo({top:0, behavior:'smooth'});
}

function router(){
  const hash = location.hash.replace('#/', '').replace('#', '');
  if(hash.startsWith('dog/')){
    renderDog(hash.slice(4));
  } else if(hash === 'matches'){
    renderMatches();
  } else {
    renderHome();
  }
}

window.addEventListener('hashchange', router);
router();
